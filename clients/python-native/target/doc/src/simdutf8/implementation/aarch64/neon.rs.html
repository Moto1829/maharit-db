85    #[inline]
86    #[allow(clippy::cast_ptr_alignment)]
87    unsafe fn load_from(ptr: *const u8) -> Self {
88        // WORKAROUND for https://github.com/rust-lang/stdarch/issues/1148
89        // The vld1q_u8 intrinsic is currently broken, it treats it as individual
90        // byte loads so the compiler sometimes decides it is a better to load
91        // individual bytes to "optimize" a subsequent SIMD shuffle
92        //
93        // This code forces a full 128-bit load.
94        let mut dst = core::mem::MaybeUninit::<uint8x16_t>::uninit();
95        core::ptr::copy_nonoverlapping(
96            ptr.cast::<u8>(),
97            dst.as_mut_ptr().cast::<u8>(),
98            core::mem::size_of::<uint8x16_t>(),
99        );
100        Self::from(dst.assume_init())
101    }

220impl Utf8CheckAlgorithm<SimdU8Value> {
221    #[inline]
222    unsafe fn must_be_2_3_continuation(prev2: SimdU8Value, prev3: SimdU8Value) -> SimdU8Value {
223        let is_third_byte = prev2.unsigned_gt(SimdU8Value::splat(0b1110_0000 - 1));
224        let is_fourth_byte = prev3.unsigned_gt(SimdU8Value::splat(0b1111_0000 - 1));
225
226        is_third_byte.or(is_fourth_byte)
227    }
228}

232unsafe fn simd_prefetch(ptr: *const u8) {
233    use core::arch::aarch64::{_prefetch, _PREFETCH_LOCALITY3, _PREFETCH_READ};

235}
