1//! Native Python extension for MaharitDB.
2//!
3//! Provides in-process graph operations without the TCP server overhead.
4//!
5//! # Usage (from Python)
6//!
7//! ```python
8//! from maharit_native import Graph
9//!
10//! g = Graph()
11//! alice = g.create_node("Person")
12//! bob   = g.create_node("Person")
13//! g.set_node_property(alice, "name", "Alice")
14//! edge  = g.create_edge(alice, bob, "KNOWS")
15//!
16//! print(g.node_count())   # 2
17//! print(g.get_node(alice))
18//! # {'id': 0, 'labels': ['Person'], 'properties': {'name': 'Alice'}}
19//! ```
20
21use maharit_core::ConcurrentGraph;
22use maharit_core::PropertyValue;
23use pyo3::exceptions::PyValueError;
24use pyo3::prelude::*;
25use pyo3::types::{PyDict, PyList};
26
27/// In-process graph database backed by lock-free DashMap shards.
28#[pyclass(name = "Graph")]
29struct PyGraph {
30    inner: ConcurrentGraph,
31}
32
33#[pymethods]
34impl PyGraph {
35    /// Create an empty graph.
36    #[new]
37    fn new() -> Self {
38        Self {
39            inner: ConcurrentGraph::new(),
40        }
41    }
42
43    /// Create a node with a single label. Returns the node ID.
44    fn create_node(&self, label: &str) -> u64 {
45        self.inner.create_node(label)
46    }
47
48    /// Create a node with multiple labels. Returns the node ID.
49    fn create_node_with_labels(&self, labels: Vec<String>) -> u64 {
50        self.inner.create_node_with_labels(labels)
51    }
52
53    /// Create a directed edge. Returns the edge ID.
54    ///
55    /// Raises `ValueError` if either node does not exist.
56    fn create_edge(&self, from: u64, to: u64, label: &str) -> PyResult<u64> {
57        self.inner
58            .create_edge(from, to, label)
59            .map_err(|e| PyValueError::new_err(e.to_string()))
60    }
61
62    /// Delete a node and all its incident edges. Returns `True` if found.
63    fn delete_node(&self, id: u64) -> bool {
64        self.inner.delete_node(id).is_some()
65    }
66
67    /// Delete an edge by ID. Returns `True` if found.
68    fn delete_edge(&self, id: u64) -> bool {
69        self.inner.delete_edge(id).is_some()
70    }
71
72    /// Set a string property on a node.
73    fn set_node_property(&self, node_id: u64, key: &str, value: &str) {
74        self.inner.set_node_property(
75            node_id,
76            key,
77            PropertyValue::String(value.to_string()),
78        );
79    }
80
81    /// Set an integer property on a node.
82    fn set_node_property_int(&self, node_id: u64, key: &str, value: i64) {
83        self.inner
84            .set_node_property(node_id, key, PropertyValue::Int(value));
85    }
86
87    /// Set a float property on a node.
88    fn set_node_property_float(&self, node_id: u64, key: &str, value: f64) {
89        self.inner
90            .set_node_property(node_id, key, PropertyValue::Float(value));
91    }
92
93    /// Set a boolean property on a node.
94    fn set_node_property_bool(&self, node_id: u64, key: &str, value: bool) {
95        self.inner
96            .set_node_property(node_id, key, PropertyValue::Bool(value));
97    }
98
99    /// Set a string property on an edge.
100    fn set_edge_property(&self, edge_id: u64, key: &str, value: &str) {
101        self.inner.set_edge_property(
102            edge_id,
103            key,
104            PropertyValue::String(value.to_string()),
105        );
106    }
107
108    /// Return a dict representation of the node, or `None` if not found.
109    fn get_node<'py>(&self, py: Python<'py>, id: u64) -> Option<Bound<'py, PyDict>> {
110        self.inner.with_node(id, |node| {
111            let d = PyDict::new(py);
112            d.set_item("id", node.id).unwrap();
113            let labels: Vec<&str> = node.labels.iter().map(|s| s.as_str()).collect();
114            d.set_item("labels", PyList::new(py, &labels).unwrap()).unwrap();
115            let props = PyDict::new(py);
116            for (k, v) in node.properties.iter() {
117                props.set_item(k, pv_to_py(py, v)).unwrap();
118            }
119            d.set_item("properties", props).unwrap();
120            d
121        })
122    }
123
124    /// Return a dict representation of the edge, or `None` if not found.
125    fn get_edge<'py>(&self, py: Python<'py>, id: u64) -> Option<Bound<'py, PyDict>> {
126        self.inner.with_edge(id, |edge| {
127            let d = PyDict::new(py);
128            d.set_item("id", edge.id).unwrap();
129            d.set_item("label", edge.label.as_str()).unwrap();
130            d.set_item("from", edge.from).unwrap();
131            d.set_item("to", edge.to).unwrap();
132            let props = PyDict::new(py);
133            for (k, v) in edge.properties.iter() {
134                props.set_item(k, pv_to_py(py, v)).unwrap();
135            }
136            d.set_item("properties", props).unwrap();
137            d
138        })
139    }
140
141    /// Outgoing edge IDs for a node.
142    fn get_outgoing_edges(&self, node_id: u64) -> Vec<u64> {
143        self.inner.get_outgoing_edges(node_id)
144    }
145
146    /// Incoming edge IDs for a node.
147    fn get_incoming_edges(&self, node_id: u64) -> Vec<u64> {
148        self.inner.get_incoming_edges(node_id)
149    }
150
151    /// Number of nodes.
152    fn node_count(&self) -> usize {
153        self.inner.node_count()
154    }
155
156    /// Number of edges.
157    fn edge_count(&self) -> usize {
158        self.inner.edge_count()
159    }
160}
161
162/// Convert a `PropertyValue` to a Python object.
163fn pv_to_py<'py>(py: Python<'py>, pv: &PropertyValue) -> Bound<'py, PyAny> {
164    match pv {
165        PropertyValue::Null => py.None().into_bound(py),
166        PropertyValue::Bool(b) => b.into_pyobject(py).unwrap().into_any(),
167        PropertyValue::Int(n) => n.into_pyobject(py).unwrap().into_any(),
168        PropertyValue::Float(f) => f.into_pyobject(py).unwrap().into_any(),
169        PropertyValue::String(s) => s.as_str().into_pyobject(py).unwrap().into_any(),
170        other => format!("{:?}", other)
171            .into_pyobject(py)
172            .unwrap()
173            .into_any(),
174    }
175}
176
177/// Register the Python module.
178#[pymodule]
179fn maharit_native(m: &Bound<'_, PyModule>) -> PyResult<()> {
180    m.add_class::<PyGraph>()?;
181    Ok(())
182}
