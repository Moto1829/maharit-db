15/// Compound consecutive tokens that have specified part-of-speech tags into a single token.
16///
17#[derive(Clone, Debug)]
18pub struct JapaneseCompoundWordTokenFilter {
19    tags: HashSet<String>,
20    #[allow(dead_code)]
21    new_tag: Option<String>,
22}
23
24impl JapaneseCompoundWordTokenFilter {
25    pub fn new(tags: HashSet<String>, new_tag: Option<String>) -> Self {
26        let tags: HashSet<String> = tags
27            .into_iter()
28            .map(|v| {
29                let mut tag_parts: Vec<&str> = v.split(',').collect();
30                tag_parts.resize(4, "*");
31                tag_parts.join(",")
32            })
33            .collect();
34
35        let new_tag = new_tag.map(|v| {
36            let mut tag_parts: Vec<&str> = v.split(',').collect();
37            tag_parts.resize(4, "*");
38            tag_parts.join(",")
39        });
40
41        Self { tags, new_tag }
42    }
43
44    pub fn from_config(config: &JapaneseCompoundWordTokenFilterConfig) -> LinderaResult<Self> {
45        let tags: HashSet<String> = config["tags"]
46            .as_array()
47            .ok_or_else(|| {
48                LinderaErrorKind::Deserialize.with_error(anyhow::anyhow!("tags is required"))
49            })?
50            .iter()
51            .map(|v| {
52                v.as_str()
53                    .ok_or_else(|| {
54                        LinderaErrorKind::Deserialize
55                            .with_error(anyhow::anyhow!("tag must be string"))
56                    })
57                    .map(|s| s.to_string())
58            })
59            .collect::<LinderaResult<HashSet<String>>>()?;
60
61        let new_tag = config
62            .get("new_tag")
63            .map(|v| {
64                v.as_str()
65                    .ok_or_else(|| {
66                        LinderaErrorKind::Deserialize
67                            .with_error(anyhow::anyhow!("new_tag must be a string"))
68                    })
69                    .map(|s| s.to_string())
70            })
71            .transpose()?;
72
73        Ok(Self::new(tags, new_tag))
74    }
75
76    // Concatenate two tokens into one.
77    fn concat_token<'a>(&self, token1: &mut Token<'a>, token2: &Token<'a>) {
78        token1.surface = Cow::Owned(format!("{}{}", token1.surface, token2.surface));
79        token1.byte_end = token2.byte_end;
80        token1.position_length += token2.position_length;
81
82        // Token details field length
83        // -4 to exclude surface, left_context_id, right_context_id and cost
84        let details_field_length = token1.dictionary.metadata.dictionary_schema.field_count() - 4;
85
86        // Make details for the new token based on the new_tag.
87        let details = match &self.new_tag {
88            Some(new_tag) => {
89                let mut details = new_tag.split(',').collect::<Vec<&str>>();
90                if details.len() < details_field_length {
91                    details.resize(details_field_length, "*");
92                } else {
93                    details.truncate(details_field_length);
94                }
95                let details: Vec<Cow<'_, str>> =
96                    details.iter().map(|s| Cow::Owned(s.to_string())).collect();
97                details
98            }
99            None => {
100                let mut details = Vec::with_capacity(details_field_length);
101                details.push(Cow::Borrowed("複合語"));
102                while details.len() < details_field_length {
103                    details.push(Cow::Borrowed("*"));
104                }
105                details
106            }
107        };
108
109        // token1.details = Some(details);
110        for (i, detail) in details.iter().enumerate() {
111            token1.set_detail(i, detail.clone());
112        }
113    }
114}

116impl TokenFilter for JapaneseCompoundWordTokenFilter {
117    fn name(&self) -> &'static str {
118        JAPANESE_COMPOUND_WORD_TOKEN_FILTER_NAME
119    }
120
121    /// Merges tokens based on matching part-of-speech tags and updates the token list.
122    ///
123    /// # Arguments
124    ///
125    /// * `tokens` - A mutable reference to a vector of tokens. The tokens will be modified in place by merging consecutive tokens that share matching tags.
126    ///
127    /// # Returns
128    ///
129    /// Returns a `LinderaResult<()>` indicating whether the operation was successful.
130    ///
131    /// # Process
132    ///
133    /// 1. **Token Processing**:
134    ///    - The function iterates over the list of tokens, and for each token, it checks the part-of-speech tags up to 4 elements (`tags_len`).
135    ///    - If the token's tag matches one of the tags specified in the configuration (`self.config.tags`), it attempts to merge the token with the subsequent tokens.
136    ///
137    /// 2. **Token Merging**:
138    ///    - When two consecutive tokens have matching tags, they are merged by concatenating their details into a single token.
139    ///    - If no matching tag is found for the next token, the current token is finalized and added to the new token list.
140    ///
141    /// 3. **Replacing Tokens**:
142    ///    - After processing all tokens, the original token list is replaced by the new list (`new_tokens`) that contains merged tokens where applicable.
143    ///
144    /// # Special Cases:
145    ///
146    /// - If no tags match, the original tokens are retained without modification.
147    /// - If multiple tokens match, they are merged into a single token.
148    ///
149    /// # Errors
150    ///
151    /// If any issue arises during token processing, the function will return an error in the form of `LinderaResult`.
152    fn apply(&self, tokens: &mut Vec<Token<'_>>) -> LinderaResult<()> {
153        // New tokens
154        let mut new_tokens = Vec::new();
155
156        // Index of the current token
157        let mut i = 0;
158
159        while i < tokens.len() {
160            // Get the current token by reference and clone it later only if needed.
161            let current = &mut tokens[i];
162            i += 1;
163
164            let current_details = current.details();
165            let current_tags_len = current_details.len().min(4);
166            let current_tag = current_details[0..current_tags_len].join(",");
167
168            // If the tag matches, merge the tokens
169            if self.tags.contains(&current_tag) {
170                // Clone the current token as it will be modified
171                let mut merged_token = current.clone();
172
173                while i < tokens.len() {
174                    let next = &mut tokens[i];
175                    let next_details = next.details();
176                    let next_tags_len = next_details.len().min(4);
177                    let next_tag = next_details[0..next_tags_len].join(",");
178
179                    // If the next tag matches, merge the tokens; otherwise, break the loop
180                    if self.tags.contains(&next_tag) {
181                        // Concatenate the current token and the next token.
182                        self.concat_token(&mut merged_token, next);
183                        i += 1; // Move to the next token
184                    } else {
185                        break; // No match, stop merging
186                    }
187                }
188                new_tokens.push(merged_token);
189            } else {
190                // No need to merge, just clone the current token
191                new_tokens.push(current.clone());
192            }
193        }
194
195        // Replace the original tokens with the new tokens after processing.
196        mem::swap(tokens, &mut new_tokens);
197
198        Ok(())
199    }
200}
