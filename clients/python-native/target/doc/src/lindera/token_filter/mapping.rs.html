13pub const MAPPING_TOKEN_FILTER_NAME: &str = "mapping";
14
15pub type MappingTokenFilterConfig = Value;
16
17/// Replace characters with the specified character mappings.
18///
19#[derive(Clone)]
20pub struct MappingTokenFilter {
21    mapping: HashMap<String, String>,
22    trie: DoubleArrayAhoCorasick<u32>,
23}
24
25impl MappingTokenFilter {
26    pub fn new(mapping: HashMap<String, String>) -> LinderaResult<Self> {
27        let mut keyset: Vec<(&[u8], u32)> = Vec::new();
28        let mut keys = mapping.keys().collect::<Vec<_>>();
29        keys.sort();
30        for (value, key) in keys.into_iter().enumerate() {
31            keyset.push((key.as_bytes(), value as u32));
32        }
33
34        let trie = DoubleArrayAhoCorasickBuilder::new()
35            .build_with_values(keyset)
36            .map_err(|err| LinderaErrorKind::Build.with_error(anyhow::anyhow!(err)))?;
37
38        Ok(Self { mapping, trie })
39    }
40
41    pub fn from_config(config: &MappingTokenFilterConfig) -> LinderaResult<Self> {
42        let mapping = config
43            .get("mapping")
44            .and_then(Value::as_object)
45            .ok_or_else(|| {
46                LinderaErrorKind::Parse.with_error(anyhow::anyhow!("mapping must be an object."))
47            })?
48            .iter()
49            .filter_map(|(k, v)| v.as_str().map(|s| (k.clone(), s.to_string())))
50            .collect::<HashMap<String, String>>();
51
52        Self::new(mapping)
53    }
54}
55
56impl TokenFilter for MappingTokenFilter {
57    fn name(&self) -> &'static str {
58        MAPPING_TOKEN_FILTER_NAME
59    }
60
61    fn apply(&self, tokens: &mut Vec<Token<'_>>) -> LinderaResult<()> {
62        for token in tokens.iter_mut() {
63            let mut result = String::new();
64            let mut start = 0_usize;
65            let len = token.surface.len();
66
67            while start < len {
68                let suffix = &token.surface[start..];
69                match self
70                    .trie
71                    .find_overlapping_iter(suffix.as_bytes())
72                    .filter(|m| m.start() == 0)
73                    .last()
74                    .map(|m| m.end())
75                {
76                    Some(prefix_len) => {
77                        let surface = &token.surface[start..start + prefix_len];
78                        let replacement = &self.mapping[surface];
79
80                        result.push_str(replacement);
81
82                        // move start offset
83                        start += prefix_len;
84                    }
85                    None => {
86                        match suffix.chars().next() {
87                            Some(c) => {
88                                result.push(c);
89
90                                // move start offset
91                                start += c.len_utf8();
92                            }
93                            None => break,
94                        }
95                    }
96                }
97            }
98
99            token.surface = Cow::Owned(result);
100        }
101
102        Ok(())
103    }
104}
