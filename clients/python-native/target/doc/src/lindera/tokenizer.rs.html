20fn yaml_to_config(file_path: &Path) -> LinderaResult<TokenizerConfig> {
21    let mut input_read = File::open(file_path).map_err(|err| {
22        LinderaErrorKind::Io.with_error(err).add_context(format!(
23            "Failed to open tokenizer config file: {}",
24            file_path.display()
25        ))
26    })?;
27
28    let mut buffer = Vec::new();
29    input_read.read_to_end(&mut buffer).map_err(|err| {
30        LinderaErrorKind::Io.with_error(err).add_context(format!(
31            "Failed to read tokenizer config file: {}",
32            file_path.display()
33        ))
34    })?;
35
36    match serde_yaml_ng::from_slice::<serde_yaml_ng::Value>(&buffer) {
37        Ok(value) => {
38            // Check if the value is a mapping.
39            match value {
40                serde_yaml_ng::Value::Mapping(_) => {
41                    Ok(serde_json::to_value(value).map_err(|err| {
42                        LinderaErrorKind::Deserialize
43                            .with_error(err)
44                            .add_context(format!(
45                                "Failed to convert YAML to JSON for config file: {}",
46                                file_path.display()
47                            ))
48                    })?)
49                }
50                _ => Err(LinderaErrorKind::Deserialize
51                    .with_error(anyhow::anyhow!("Invalid YAML"))
52                    .add_context(format!(
53                        "Config file must contain a YAML mapping: {}",
54                        file_path.display()
55                    ))),
56            }
57        }
58        Err(err) => Err(LinderaErrorKind::Deserialize
59            .with_error(err)
60            .add_context(format!(
61                "Failed to parse YAML config file: {}",
62                file_path.display()
63            ))),
64    }
65}

78    if config.get("segmenter").is_none() {
79        config["segmenter"] = json!({});
80    }
81
82    if config.get("character_filters").is_none() {
83        config["character_filters"] = json!([]);
84    }
85
86    if config.get("token_filters").is_none() {
87        config["token_filters"] = json!([]);

165    /// Segmenter
166    /// The `segmenter` field is an instance of the `Segmenter` struct, which is responsible for
167    /// segmenting text into tokens. This is a core component of the tokenizer, enabling it to
168    /// break down input text into manageable and meaningful units for further processing.
169    pub segmenter: Segmenter,
170
171    /// Character filters
172    /// A vector of boxed character filters that will be applied to the input text
173    /// before tokenization. Each character filter is responsible for transforming
174    /// the input text in a specific way, such as normalizing characters or removing
175    /// unwanted characters.
176    pub character_filters: Vec<BoxCharacterFilter>,
177
178    /// Token filters
179    /// A vector of boxed token filters that will be applied to the tokens during tokenization.
180    /// Each token filter is a boxed trait object implementing the `TokenFilter` trait, allowing
181    /// for various transformations and processing steps to be applied to the tokens.
182    pub token_filters: Vec<BoxTokenFilter>,
183}

186    /// Creates a new `Tokenizer` instance from a provided `Segmenter`.
187    ///
188    /// # Arguments
189    ///
190    /// * `segmenter` - An instance of the `Segmenter` struct, which is responsible for the core tokenization process.
191    ///
192    /// # Returns
193    ///
194    /// Returns a new `Tokenizer` instance that uses the provided `segmenter` for tokenization, with empty character and token filters.
195    ///
196    /// # Details
197    ///
198    /// - `segmenter`: The segmenter is responsible for handling the actual segmentation and tokenization of text. It is passed into the `Tokenizer` during initialization.
199    /// - `character_filters`: This is initialized as an empty vector and can be modified later to include character filters.
200    /// - `token_filters`: This is also initialized as an empty vector and can be modified later to include token filters.
201    pub fn new(segmenter: Segmenter) -> Self {
202        Self {
203            segmenter,
204            character_filters: Vec::new(),
205            token_filters: Vec::new(),
206        }
207    }

209    pub fn from_config(config: &TokenizerConfig) -> LinderaResult<Self> {
210        let segmenter_config = config.get("segmenter").ok_or_else(|| {
211            LinderaErrorKind::Deserialize.with_error(anyhow::anyhow!("missing segmenter config."))
212        })?;
213        let segmenter = Segmenter::from_config(segmenter_config)?;
214
215        // Create a tokenizer from the segmenter.
216        let mut tokenizer = Tokenizer::new(segmenter);
217
218        // Load character filter settings from the tokenizer config if it is not empty.
219        if let Some(character_filter_settings) = config["character_filters"].as_array() {
220            for character_filter_setting in character_filter_settings {
221                let character_filter_name = character_filter_setting["kind"].as_str();
222                if let Some(character_filter_name) = character_filter_name {
223                    // Append a character filter to the tokenizer.
224                    tokenizer.append_character_filter(CharacterFilterLoader::load_from_value(
225                        character_filter_name,
226                        &character_filter_setting["args"],
227                    )?);
228                }
229            }
230        }
231
232        // Load token filter settings from the tokenizer config if it is not empty.
233        if let Some(token_filter_settings) = config["token_filters"].as_array() {
234            for token_filter_setting in token_filter_settings {
235                let token_filter_name = token_filter_setting["kind"].as_str();
236                if let Some(token_filter_name) = token_filter_name {
237                    // Append a token filter to the tokenizer.
238                    tokenizer.append_token_filter(TokenFilterLoader::load_from_value(
239                        token_filter_name,
240                        &token_filter_setting["args"],
241                    )?);
242                }
243            }
244        }
245
246        Ok(tokenizer)
247    }

289    /// Tokenizes the input text using the tokenizer's segmenter, character filters, and token filters.
290    ///
291    /// # Arguments
292    ///
293    /// * `text` - A reference to the input text (`&str`) that will be tokenized.
294    ///
295    /// # Returns
296    ///
297    /// Returns a `LinderaResult` containing a vector of `Token`s, where each `Token` represents a segment of the tokenized text.
298    ///
299    /// # Process
300    ///
301    /// 1. **Apply character filters**:
302    ///    - If any character filters are defined, they are applied to the input text before tokenization.
303    ///    - The `offsets`, `diffs`, and `text_len` are recorded for each character filter.
304    /// 2. **Segment the text**:
305    ///    - The `segmenter` divides the (potentially filtered) text into tokens.
306    /// 3. **Apply token filters**:
307    ///    - If any token filters are defined, they are applied to the segmented tokens.
308    /// 4. **Correct token offsets**:
309    ///    - If character filters were applied, the byte offsets of each token are corrected to account for changes introduced by those filters.
310    ///
311    /// # Errors
312    ///
313    /// - Returns an error if any of the character or token filters fail during processing.
314    /// - Returns an error if the segmentation process fails.
315    ///
316    /// # Details
317    ///
318    /// - `Cow<'a, str>` is used for the `normalized_text`, allowing the function to either borrow the original text or create an owned version if the text needs modification.
319    /// - If no character filters are applied, the original `text` is used as-is for segmentation.
320    /// - Token offsets are adjusted after the tokenization process if character filters were applied to ensure the byte positions of each token are accurate relative to the original text.
321    pub fn tokenize<'a>(&'a self, text: &'a str) -> LinderaResult<Vec<Token<'a>>> {
322        let mut lattice = Lattice::default();
323        self.tokenize_with_lattice(text, &mut lattice)
324    }

326    /// Tokenizes the input text using the tokenizer's segmenter, character filters, and token filters.
327    ///
328    /// # Arguments
329    ///
330    /// * `text` - A reference to the input text (`&str`) that will be tokenized.
331    /// * `lattice` - A mutable reference to a `Lattice` structure. This allows reusing the lattice across multiple calls to avoid memory allocation.
332    ///
333    /// # Returns
334    ///
335    /// Returns a `LinderaResult` containing a vector of `Token`s, where each `Token` represents a segment of the tokenized text.
336    ///
337    /// # Process
338    ///
339    /// 1. **Apply character filters**:
340    ///    - If any character filters are defined, they are applied to the input text before tokenization.
341    ///    - The `offsets`, `diffs`, and `text_len` are recorded for each character filter.
342    /// 2. **Segment the text**:
343    ///    - The `segmenter` divides the (potentially filtered) text into tokens.
344    /// 3. **Apply token filters**:
345    ///    - If any token filters are defined, they are applied to the segmented tokens.
346    /// 4. **Correct token offsets**:
347    ///    - If character filters were applied, the byte offsets of each token are corrected to account for changes introduced by those filters.
348    ///
349    /// # Errors
350    ///
351    /// - Returns an error if any of the character or token filters fail during processing.
352    /// - Returns an error if the segmentation process fails.
353    ///
354    /// # Details
355    ///
356    /// - `Cow<'a, str>` is used for the `normalized_text`, allowing the function to either borrow the original text or create an owned version if the text needs modification.
357    /// - If no character filters are applied, the original `text` is used as-is for segmentation.
358    /// - Token offsets are adjusted after the tokenization process if character filters were applied to ensure the byte positions of each token are accurate relative to the original text.
359    pub fn tokenize_with_lattice<'a>(
360        &'a self,
361        text: &'a str,
362        lattice: &mut Lattice,
363    ) -> LinderaResult<Vec<Token<'a>>> {
364        let mut normalized_text: Cow<'a, str> = Cow::Borrowed(text);
365
366        let mut offset_mappings: Vec<OffsetMapping> =
367            Vec::with_capacity(self.character_filters.len());
368
369        // Apply character filters to the text if it is not empty.
370        // Optimize: Only convert to mutable when we have filters to apply
371        if !self.character_filters.is_empty() {
372            // Convert to owned string once for all filters
373            let text_mut = normalized_text.to_mut();
374
375            for character_filter in &self.character_filters {
376                let mapping = character_filter.apply(text_mut)?;
377
378                if !mapping.is_empty() {
379                    // Record the offset mapping of each character filter in reverse order
380                    // since we need to apply corrections in reverse order
381                    offset_mappings.push(mapping);
382                }
383            }
384        }
385
386        // Store the final text length for offset correction
387        let final_text_len = normalized_text.len();
388
389        // Segment a text.
390        // Segment a text.
391        let mut tokens = self
392            .segmenter
393            .segment_with_lattice(normalized_text, lattice)?;
394
395        // Apply token filters to the tokens if they are not empty.
396        for token_filter in &self.token_filters {
397            token_filter.apply(&mut tokens)?;
398        }
399
400        // Correct token offsets if character filters are applied.
401        // Apply corrections in reverse order (last filter first)
402        if !offset_mappings.is_empty() {
403            for token in tokens.iter_mut() {
404                // Apply corrections in reverse order to undo the transformations
405                for mapping in offset_mappings.iter().rev() {
406                    // Override start.
407                    token.byte_start = mapping.correct_offset(token.byte_start, final_text_len);
408                    // Override end.
409                    token.byte_end = mapping.correct_offset(token.byte_end, final_text_len);
410                }
411            }
412        }
413
414        Ok(tokens)
415    }

417    /// Tokenizes the input text and returns the top-N results.
418    ///
419    /// Each result is a `Vec<Token>` with character/token filters applied.
420    /// Results are ordered by cost (best first).
421    pub fn tokenize_nbest<'a>(
422        &'a self,
423        text: &'a str,
424        n: usize,
425        unique: bool,
426        cost_threshold: Option<i64>,
427    ) -> LinderaResult<Vec<(Vec<Token<'a>>, i64)>> {
428        let mut lattice = Lattice::default();
429        self.tokenize_nbest_with_lattice(text, &mut lattice, n, unique, cost_threshold)
430    }
431
432    /// Tokenizes the input text and returns the top-N results with costs.
433    /// Each result is a (tokens, cost) pair.
434    /// If `unique` is true, results with the same word boundaries are deduplicated.
435    /// If `cost_threshold` is Some(t), paths whose cost exceeds best_cost + t
436    /// are discarded.
437    pub fn tokenize_nbest_with_lattice<'a>(
438        &'a self,
439        text: &'a str,
440        lattice: &mut Lattice,
441        n: usize,
442        unique: bool,
443        cost_threshold: Option<i64>,
444    ) -> LinderaResult<Vec<(Vec<Token<'a>>, i64)>> {
445        let mut normalized_text: Cow<'a, str> = Cow::Borrowed(text);
446
447        let mut offset_mappings: Vec<OffsetMapping> =
448            Vec::with_capacity(self.character_filters.len());
449
450        if !self.character_filters.is_empty() {
451            let text_mut = normalized_text.to_mut();
452            for character_filter in &self.character_filters {
453                let mapping = character_filter.apply(text_mut)?;
454                if !mapping.is_empty() {
455                    offset_mappings.push(mapping);
456                }
457            }
458        }
459
460        let final_text_len = normalized_text.len();
461
462        let mut all_results = self.segmenter.segment_nbest_with_lattice(
463            normalized_text,
464            lattice,
465            n,
466            unique,
467            cost_threshold,
468        )?;
469
470        // Apply token filters and offset corrections to each result
471        for (tokens, _cost) in &mut all_results {
472            for token_filter in &self.token_filters {
473                token_filter.apply(tokens)?;
474            }
475
476            if !offset_mappings.is_empty() {
477                for token in tokens.iter_mut() {
478                    for mapping in offset_mappings.iter().rev() {
479                        token.byte_start = mapping.correct_offset(token.byte_start, final_text_len);
480                        token.byte_end = mapping.correct_offset(token.byte_end, final_text_len);
481                    }
482                }
483            }
484        }
485
486        Ok(all_results)
487    }

490impl Clone for Tokenizer {
491    /// Creates a deep clone of the `Tokenizer` instance, including all character filters, token filters, and the segmenter.
492    ///
493    /// # Returns
494    ///
495    /// Returns a new `Tokenizer` instance that is a deep clone of the current instance. All internal filters and the segmenter are cloned.
496    ///
497    /// # Details
498    ///
499    /// - **Character Filters**: Each character filter is cloned by calling its `box_clone` method, which ensures that any dynamically dispatched filters are properly cloned.
500    /// - **Token Filters**: Similarly, each token filter is cloned using the `box_clone` method to handle dynamic dispatch.
501    /// - **Segmenter**: The segmenter is cloned using its `clone` method.
502    ///
503    /// # Notes
504    ///
505    /// - This method performs deep cloning, meaning that all internal filters and segmenter instances are fully duplicated.
506    /// - The `box_clone` method is used to clone the dynamically dispatched filter objects (`BoxCharacterFilter` and `BoxTokenFilter`).
507    fn clone(&self) -> Self {
508        let character_filters: Vec<BoxCharacterFilter> = self
509            .character_filters
510            .iter()
511            .map(|filter| filter.box_clone())
512            .collect();
513
514        let token_filters: Vec<BoxTokenFilter> = self
515            .token_filters
516            .iter()
517            .map(|filter| filter.box_clone())
518            .collect();
519
520        Tokenizer {
521            character_filters,
522            segmenter: self.segmenter.clone(),
523            token_filters,
524        }
525    }
526}

891    #[cfg(feature = "embed-ipadic")]
892    fn test_tokenize_nbest_1best_matches_tokenize() {
893        use crate::tokenizer::TokenizerBuilder;
894
895        let mut builder = TokenizerBuilder::new().unwrap();
896        builder.set_segmenter_dictionary("embedded://ipadic");
897
898        let tokenizer = builder.build().unwrap();
899
900        let text = "すもももももももものうち";
901        let normal_tokens = tokenizer.tokenize(text).unwrap();
902        let nbest_results = tokenizer.tokenize_nbest(text, 1, false, None).unwrap();
903
904        assert_eq!(nbest_results.len(), 1);
905        let (nbest_tokens, _cost) = &nbest_results[0];
906        assert_eq!(normal_tokens.len(), nbest_tokens.len());
907        for (normal, nbest) in normal_tokens.iter().zip(nbest_tokens.iter()) {
908            assert_eq!(normal.surface.as_ref(), nbest.surface.as_ref());
909        }
910    }
