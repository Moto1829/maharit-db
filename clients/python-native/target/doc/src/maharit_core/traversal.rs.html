6/// トラバーサルの方向
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub enum Direction {
9    /// 出るエッジを辿る
10    Outgoing,
11    /// 入るエッジを辿る
12    Incoming,
13    /// 両方向
14    Both,
15}

17/// Type alias for edge filter closures used in traversal
18type EdgeFilter<'a> = Option<Box<dyn Fn(&Edge) -> bool + 'a>>;
19/// Type alias for node filter closures used in traversal
20type NodeFilter<'a> = Option<Box<dyn Fn(&Node) -> bool + 'a>>;

22/// トラバーサルビルダー
23pub struct Traversal<'a> {
24    graph: &'a Graph,
25    start: NodeId,
26    direction: Direction,
27    max_depth: Option<usize>,
28    edge_filter: EdgeFilter<'a>,
29    node_filter: NodeFilter<'a>,
30}
31
32impl<'a> Traversal<'a> {
33    pub fn new(graph: &'a Graph, start: NodeId) -> Self {
34        Self {
35            graph,
36            start,
37            direction: Direction::Outgoing,
38            max_depth: None,
39            edge_filter: None,
40            node_filter: None,
41        }
42    }
43
44    /// 探索方向を設定
45    pub fn direction(mut self, direction: Direction) -> Self {
46        self.direction = direction;
47        self
48    }
49
50    /// 最大深さを設定
51    pub fn max_depth(mut self, depth: usize) -> Self {
52        self.max_depth = Some(depth);
53        self
54    }
55
56    /// エッジのフィルタを設定
57    pub fn with_edge_filter<F>(mut self, filter: F) -> Self
58    where
59        F: Fn(&Edge) -> bool + 'a,
60    {
61        self.edge_filter = Some(Box::new(filter));
62        self
63    }
64
65    /// ノードのフィルタを設定
66    pub fn with_node_filter<F>(mut self, filter: F) -> Self
67    where
68        F: Fn(&Node) -> bool + 'a,
69    {
70        self.node_filter = Some(Box::new(filter));
71        self
72    }
73
74    /// BFS（幅優先探索）でイテレート
75    pub fn bfs(self) -> BfsIterator<'a> {
76        BfsIterator::new(self)
77    }
78
79    /// DFS（深さ優先探索）でイテレート
80    pub fn dfs(self) -> DfsIterator<'a> {
81        DfsIterator::new(self)
82    }
83
84    /// Push filtered neighbors of `node_id` directly into the provided `push` closure.
85    /// Avoids allocating a Vec by inlining the edge traversal and filter logic.
86    fn push_neighbors<F>(&self, node_id: NodeId, mut push: F)
87    where
88        F: FnMut(NodeId),
89    {
90        let graph = &self.graph;
91        let edge_filter = &self.edge_filter;
92
93        if self.direction == Direction::Outgoing || self.direction == Direction::Both {
94            for edge in graph.get_outgoing_edges(node_id) {
95                if edge_filter.as_ref().is_none_or(|f| f(edge)) {
96                    push(edge.to);
97                }
98            }
99        }
100
101        if self.direction == Direction::Incoming || self.direction == Direction::Both {
102            for edge in graph.get_incoming_edges(node_id) {
103                if edge_filter.as_ref().is_none_or(|f| f(edge)) {
104                    push(edge.from);
105                }
106            }
107        }
108    }
109
110    fn should_visit_node(&self, node: &Node) -> bool {
111        self.node_filter.as_ref().is_none_or(|f| f(node))
112    }
113}

115/// BFSイテレータ
116pub struct BfsIterator<'a> {
117    traversal: Traversal<'a>,
118    queue: VecDeque<(NodeId, usize)>, // (node_id, depth)
119    visited: HashSet<NodeId>,
120}
121
122impl<'a> BfsIterator<'a> {
123    fn new(traversal: Traversal<'a>) -> Self {
124        let mut queue = VecDeque::new();
125        let mut visited = HashSet::new();
126
127        queue.push_back((traversal.start, 0));
128        visited.insert(traversal.start);
129
130        Self {
131            traversal,
132            queue,
133            visited,
134        }
135    }
136}
137
138impl<'a> Iterator for BfsIterator<'a> {
139    type Item = (NodeId, usize); // (node_id, depth)
140
141    fn next(&mut self) -> Option<Self::Item> {
142        while let Some((node_id, depth)) = self.queue.pop_front() {
143            // 深さ制限のチェック
144            if let Some(max_depth) = self.traversal.max_depth
145                && depth > max_depth
146            {
147                continue;
148            }
149
150            // ノードフィルタのチェック
151            if let Some(node) = self.traversal.graph.get_node(node_id) {
152                if !self.traversal.should_visit_node(node) {
153                    continue;
154                }
155            } else {
156                continue;
157            }
158
159            // 隣接ノードをキューに追加（Vec アロケーションなし）
160            if self.traversal.max_depth.is_none_or(|max| depth < max) {
161                self.traversal.push_neighbors(node_id, |neighbor_id| {
162                    if !self.visited.contains(&neighbor_id) {
163                        self.visited.insert(neighbor_id);
164                        self.queue.push_back((neighbor_id, depth + 1));
165                    }
166                });
167            }
168
169            return Some((node_id, depth));
170        }
171
172        None
173    }
174}

257/// 2ノード間の最短パス（ホップ数）を探索
258pub fn shortest_path(graph: &Graph, from: NodeId, to: NodeId) -> Option<Path> {
259    if from == to {
260        return Some(Path::new(vec![from]));
261    }
262
263    let mut queue = VecDeque::new();
264    let mut visited = HashSet::new();
265    let mut parent: std::collections::HashMap<NodeId, NodeId> = std::collections::HashMap::new();
266
267    queue.push_back(from);
268    visited.insert(from);
269
270    while let Some(current) = queue.pop_front() {
271        for edge in graph.get_outgoing_edges(current) {
272            let neighbor = edge.to;
273            if !visited.contains(&neighbor) {
274                visited.insert(neighbor);
275                parent.insert(neighbor, current);
276
277                if neighbor == to {
278                    // パスを復元
279                    let mut path = vec![to];
280                    let mut node = to;
281                    while let Some(&p) = parent.get(&node) {
282                        path.push(p);
283                        node = p;
284                    }
285                    path.reverse();
286                    return Some(Path::new(path));
287                }
288
289                queue.push_back(neighbor);
290            }
291        }
292    }
293
294    None
295}

297/// 2ノード間の全最短パスを探索
298///
299/// BFSで各ノードの全前任者(predecessors)を記録し、
300/// ゴールから再帰的に全パスを復元する。
301pub fn all_shortest_paths(graph: &Graph, from: NodeId, to: NodeId) -> Vec<Path> {
302    if from == to {
303        return vec![Path::new(vec![from])];
304    }
305
306    let mut queue = VecDeque::new();
307    let mut distances: HashMap<NodeId, usize> = HashMap::new();
308    let mut predecessors: HashMap<NodeId, Vec<NodeId>> = HashMap::new();
309
310    queue.push_back(from);
311    distances.insert(from, 0);
312
313    let mut target_distance: Option<usize> = None;
314
315    while let Some(current) = queue.pop_front() {
316        let current_dist = distances[&current];
317
318        // If we've found paths to target and current distance exceeds target distance, stop
319        if let Some(td) = target_distance
320            && current_dist >= td
321        {
322            continue;
323        }
324
325        for edge in graph.get_outgoing_edges(current) {
326            let neighbor = edge.to;
327            let new_dist = current_dist + 1;
328
329            match distances.get(&neighbor) {
330                None => {
331                    // First time visiting this node
332                    distances.insert(neighbor, new_dist);
333                    predecessors.insert(neighbor, vec![current]);
334                    queue.push_back(neighbor);
335
336                    if neighbor == to {
337                        target_distance = Some(new_dist);
338                    }
339                }
340                Some(&existing_dist) if existing_dist == new_dist => {
341                    // Same distance path found, add predecessor
342                    predecessors.get_mut(&neighbor).unwrap().push(current);
343                }
344                _ => {
345                    // Longer path, ignore
346                }
347            }
348        }
349    }
350
351    // Reconstruct all paths from predecessors
352    if !predecessors.contains_key(&to) && from != to {
353        return vec![];
354    }
355
356    let mut all_paths = Vec::new();
357    reconstruct_all_paths(from, to, &predecessors, &mut vec![to], &mut all_paths);
358    all_paths
359}
360
361/// Helper function to recursively reconstruct all paths from predecessors
362fn reconstruct_all_paths(
363    from: NodeId,
364    current: NodeId,
365    predecessors: &HashMap<NodeId, Vec<NodeId>>,
366    current_path: &mut Vec<NodeId>,
367    all_paths: &mut Vec<Path>,
368) {
369    if current == from {
370        let mut path = current_path.clone();
371        path.reverse();
372        all_paths.push(Path::new(path));
373        return;
374    }
375
376    if let Some(preds) = predecessors.get(&current) {
377        for &pred in preds {
378            current_path.push(pred);
379            reconstruct_all_paths(from, pred, predecessors, current_path, all_paths);
380            current_path.pop();
381        }
382    }
383}

385/// 2ノード間の全単純パスを列挙（DFSバックトラッキング）
386///
387/// max_depth: 最大ホップ数の上限（Noneで無制限）
388/// 注意: グラフが大きい場合は指数的に増加する可能性がある
389pub fn all_paths(graph: &Graph, from: NodeId, to: NodeId, max_depth: Option<usize>) -> Vec<Path> {
390    let mut results = Vec::new();
391    if graph.get_node(from).is_none() || graph.get_node(to).is_none() {
392        return results;
393    }
394    if from == to {
395        return vec![Path::new(vec![from])];
396    }
397    let mut visited = HashSet::new();
398    visited.insert(from);
399    let mut current_path = vec![from];
400    all_paths_dfs(
401        graph,
402        from,
403        to,
404        max_depth,
405        0,
406        &mut visited,
407        &mut current_path,
408        &mut results,
409    );
410    results
411}
412
413#[allow(clippy::too_many_arguments)]
414fn all_paths_dfs(
415    graph: &Graph,
416    current: NodeId,
417    to: NodeId,
418    max_depth: Option<usize>,
419    depth: usize,
420    visited: &mut HashSet<NodeId>,
421    current_path: &mut Vec<NodeId>,
422    results: &mut Vec<Path>,
423) {
424    if let Some(max) = max_depth
425        && depth >= max
426    {
427        return;
428    }
429    for edge in graph.get_outgoing_edges(current) {
430        let neighbor = edge.to;
431        if neighbor == to {
432            let mut path = current_path.clone();
433            path.push(to);
434            results.push(Path::new(path));
435        } else if !visited.contains(&neighbor) {
436            visited.insert(neighbor);
437            current_path.push(neighbor);
438            all_paths_dfs(
439                graph,
440                neighbor,
441                to,
442                max_depth,
443                depth + 1,
444                visited,
445                current_path,
446                results,
447            );
448            current_path.pop();
449            visited.remove(&neighbor);
450        }
451    }
452}

454/// 2ノード間にパスが存在するか確認
455pub fn has_path(graph: &Graph, from: NodeId, to: NodeId) -> bool {
456    if from == to {
457        return true;
458    }
459
460    let mut queue = VecDeque::new();
461    let mut visited = HashSet::new();
462
463    queue.push_back(from);
464    visited.insert(from);
465
466    while let Some(current) = queue.pop_front() {
467        for edge in graph.get_outgoing_edges(current) {
468            let neighbor = edge.to;
469            if neighbor == to {
470                return true;
471            }
472            if !visited.contains(&neighbor) {
473                visited.insert(neighbor);
474                queue.push_back(neighbor);
475            }
476        }
477    }
478
479    false
480}

533/// 重み取得関数の型
534pub type WeightFn<'a> = Box<dyn Fn(&Edge) -> f64 + 'a>;

536/// Dijkstra法による最短経路探索
537pub struct Dijkstra<'a> {
538    graph: &'a Graph,
539    weight_fn: WeightFn<'a>,
540}
541
542impl<'a> Dijkstra<'a> {
543    /// デフォルトの重み（すべて1.0）で作成
544    pub fn new(graph: &'a Graph) -> Self {
545        Self {
546            graph,
547            weight_fn: Box::new(|_| 1.0),
548        }
549    }
550
551    /// エッジプロパティから重みを取得
552    pub fn with_weight_property(graph: &'a Graph, property: &'a str) -> Self {
553        Self {
554            graph,
555            weight_fn: Box::new(move |edge| {
556                edge.get_property(property)
557                    .and_then(|v| match v {
558                        PropertyValue::Int(n) => Some(*n as f64),
559                        PropertyValue::Float(n) => Some(*n),
560                        _ => None,
561                    })
562                    .unwrap_or(1.0)
563            }),
564        }
565    }
566
567    /// カスタム重み関数を使用
568    pub fn with_weight_fn<F>(graph: &'a Graph, weight_fn: F) -> Self
569    where
570        F: Fn(&Edge) -> f64 + 'a,
571    {
572        Self {
573            graph,
574            weight_fn: Box::new(weight_fn),
575        }
576    }
577
578    /// 単一始点から特定の終点への最短経路
579    pub fn find_path(&self, from: NodeId, to: NodeId) -> Option<WeightedPath> {
580        if from == to {
581            return Some(WeightedPath::new(vec![from], vec![], 0.0));
582        }
583
584        // Dense Vec indexed by NodeId for O(1) access (no hash computation)
585        let n = self.graph.node_capacity();
586        let mut distances = vec![f64::INFINITY; n];
587        let mut previous: Vec<Option<(NodeId, EdgeId)>> = vec![None; n];
588        let mut heap = BinaryHeap::new();
589
590        distances[from as usize] = 0.0;
591        heap.push(DijkstraEntry {
592            node: from,
593            distance: 0.0,
594        });
595
596        while let Some(DijkstraEntry { node, distance }) = heap.pop() {
597            if node == to {
598                return Some(self.reconstruct_path(from, to, &previous, distance));
599            }
600
601            // すでにより短い経路が見つかっている場合はスキップ
602            if distance > distances[node as usize] {
603                continue;
604            }
605
606            for edge in self.graph.get_outgoing_edges(node) {
607                let weight = (self.weight_fn)(edge);
608                let next_distance = distance + weight;
609                let neighbor = edge.to;
610
611                if next_distance < distances[neighbor as usize] {
612                    distances[neighbor as usize] = next_distance;
613                    previous[neighbor as usize] = Some((node, edge.id));
614                    heap.push(DijkstraEntry {
615                        node: neighbor,
616                        distance: next_distance,
617                    });
618                }
619            }
620        }
621
622        None
623    }
624
625    /// 単一始点から全ノードへの最短距離
626    ///
627    /// 内部で稠密 Vec を使い、ハッシュ計算を排除する。
628    pub fn distances_from(&self, from: NodeId) -> HashMap<NodeId, f64> {
629        let n = self.graph.node_capacity();
630        let mut dist_vec = vec![f64::INFINITY; n];
631        let mut heap = BinaryHeap::new();
632
633        if (from as usize) < n {
634            dist_vec[from as usize] = 0.0;
635        }
636        heap.push(DijkstraEntry {
637            node: from,
638            distance: 0.0,
639        });
640
641        while let Some(DijkstraEntry { node, distance }) = heap.pop() {
642            let node_idx = node as usize;
643            if node_idx >= n || distance > dist_vec[node_idx] {
644                continue;
645            }
646
647            for edge in self.graph.get_outgoing_edges(node) {
648                let weight = (self.weight_fn)(edge);
649                let next_distance = distance + weight;
650                let neighbor = edge.to;
651                let neighbor_idx = neighbor as usize;
652
653                if neighbor_idx < n && next_distance < dist_vec[neighbor_idx] {
654                    dist_vec[neighbor_idx] = next_distance;
655                    heap.push(DijkstraEntry {
656                        node: neighbor,
657                        distance: next_distance,
658                    });
659                }
660            }
661        }
662
663        // Collect finite distances into HashMap for the public API
664        self.graph
665            .nodes()
666            .map(|n| n.id)
667            .filter_map(|id| {
668                let d = dist_vec[id as usize];
669                if d.is_finite() { Some((id, d)) } else { None }
670            })
671            .collect()
672    }
673
674    fn reconstruct_path(
675        &self,
676        from: NodeId,
677        to: NodeId,
678        previous: &[Option<(NodeId, EdgeId)>],
679        total_weight: f64,
680    ) -> WeightedPath {
681        let mut nodes = vec![to];
682        let mut edges = Vec::new();
683        let mut current = to;
684
685        while current != from {
686            if let Some((prev_node, edge_id)) =
687                previous.get(current as usize).and_then(|v| *v)
688            {
689                nodes.push(prev_node);
690                edges.push(edge_id);
691                current = prev_node;
692            } else {
693                break;
694            }
695        }
696
697        nodes.reverse();
698        edges.reverse();
699
700        WeightedPath::new(nodes, edges, total_weight)
701    }
702}

749        H: Fn(NodeId, NodeId) -> f64 + 'a,
750    {
751        Self {
752            graph,
753            weight_fn: Box::new(|_| 1.0),
754            heuristic: Box::new(heuristic),
755        }
756    }

758    /// 重み関数も指定
759    pub fn with_weight_fn<W, H>(graph: &'a Graph, weight_fn: W, heuristic: H) -> Self
760    where
761        W: Fn(&Edge) -> f64 + 'a,
762        H: Fn(NodeId, NodeId) -> f64 + 'a,
763    {
764        Self {
765            graph,
766            weight_fn: Box::new(weight_fn),
767            heuristic: Box::new(heuristic),
768        }
769    }
770
771    /// A*で最短経路を探索
772    pub fn find_path(&self, from: NodeId, to: NodeId) -> Option<WeightedPath> {
773        if from == to {
774            return Some(WeightedPath::new(vec![from], vec![], 0.0));
775        }
776
777        let mut g_scores: HashMap<NodeId, f64> = HashMap::new();
778        let mut previous: HashMap<NodeId, (NodeId, EdgeId)> = HashMap::new();
779        let mut heap = BinaryHeap::new();
780        let mut closed: HashSet<NodeId> = HashSet::new();
781
782        g_scores.insert(from, 0.0);
783        heap.push(AStarEntry {
784            node: from,
785            g_score: 0.0,
786            f_score: (self.heuristic)(from, to),
787        });
788
789        while let Some(AStarEntry { node, g_score, .. }) = heap.pop() {
790            if node == to {
791                return Some(self.reconstruct_path(from, to, &previous, g_score));
792            }
793
794            if closed.contains(&node) {
795                continue;
796            }
797            closed.insert(node);
798
799            for edge in self.graph.get_outgoing_edges(node) {
800                let neighbor = edge.to;
801                if closed.contains(&neighbor) {
802                    continue;
803                }
804
805                let weight = (self.weight_fn)(edge);
806                let tentative_g = g_score + weight;
807
808                if !g_scores.contains_key(&neighbor) || tentative_g < g_scores[&neighbor] {
809                    g_scores.insert(neighbor, tentative_g);
810                    previous.insert(neighbor, (node, edge.id));
811
812                    let f_score = tentative_g + (self.heuristic)(neighbor, to);
813                    heap.push(AStarEntry {
814                        node: neighbor,
815                        g_score: tentative_g,
816                        f_score,
817                    });
818                }
819            }
820        }
821
822        None
823    }
824
825    fn reconstruct_path(
826        &self,
827        from: NodeId,
828        to: NodeId,
829        previous: &HashMap<NodeId, (NodeId, EdgeId)>,
830        total_weight: f64,
831    ) -> WeightedPath {
832        let mut nodes = vec![to];
833        let mut edges = Vec::new();
834        let mut current = to;
835
836        while current != from {
837            if let Some(&(prev_node, edge_id)) = previous.get(&current) {
838                nodes.push(prev_node);
839                edges.push(edge_id);
840                current = prev_node;
841            } else {
842                break;
843            }
844        }
845
846        nodes.reverse();
847        edges.reverse();
848
849        WeightedPath::new(nodes, edges, total_weight)
850    }
